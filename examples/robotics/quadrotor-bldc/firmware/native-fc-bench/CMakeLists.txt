cmake_minimum_required(VERSION 3.13)
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Native RP2354B Arm target")
set(PICO_BOARD openfc_lite_rp2354b_bench CACHE STRING "Native OpenFC board")
list(APPEND PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/boards")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(openfc_motor_bench C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
# Append reproducibility flags after the SDK toolchain establishes Cortex-M33.
# Keep SDK assertions enabled in this physical commissioning firmware.
add_compile_options(-UNDEBUG "-ffile-prefix-map=${CMAKE_CURRENT_LIST_DIR}=/src/native-fc-bench")
if(DEFINED BENCH_REPRO_ROOT)
  add_compile_options("-ffile-prefix-map=${BENCH_REPRO_ROOT}=/build/native-fc-bench")
endif()
pico_sdk_init()
add_executable(openfc_motor_bench src/main.c src/protocol.c)
pico_generate_pio_header(openfc_motor_bench ${CMAKE_CURRENT_LIST_DIR}/src/dshot.pio)
target_include_directories(openfc_motor_bench PRIVATE src)
target_compile_options(openfc_motor_bench PRIVATE -Wall -Wextra -Werror)
target_link_libraries(openfc_motor_bench pico_stdlib hardware_pio hardware_dma hardware_uart hardware_clocks)
pico_enable_stdio_uart(openfc_motor_bench 1)
pico_enable_stdio_usb(openfc_motor_bench 0)
pico_set_program_name(openfc_motor_bench "openfc-native-motor-bench")
pico_set_program_version(openfc_motor_bench "1.0.0")
pico_add_extra_outputs(openfc_motor_bench)
